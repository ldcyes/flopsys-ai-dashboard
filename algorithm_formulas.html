def reference_weight_bytes(model, cin, cout, *, family=None, role=None, tp=1, shard="replicated"):
    """Exact most-loaded-rank tensor bytes, including each 2D scale tail."""
    if shard not in {"replicated", "column", "row"}:
        raise ValueError(f"Unknown reference weight sharding: {shard}")
    cin, cout, tp = int(cin), int(cout), int(tp)
    if min(cin, cout, tp) < 1:
        raise ValueError("Reference weight dimensions and TP must be positive")
    if shard == "column":
        cout = ceil(cout / tp)
    elif shard == "row":
        cin = ceil(cin / tp)
    policy = resolve_reference_precision(model, family=family, role=role)
    return cin * cout * policy["weight_bpe"] + ceil(cin / 128) * ceil(cout / 128) * policy["weight_scale_bpe"]

def _cycle_counts(remaining, cap, probabilities):
    """Expected visits to each query prefix, with finite-output clipping."""
    visits = [0.0] * (remaining + 1)
    visits[remaining] = 1.0
    counts = [0.0] * 6
    for left in range(remaining, 0, -1):
        k = min(cap, left)
        counts[k] += visits[left]
        survival = 1.0
        for accepted in range(k):
            probability = survival * (1 - probabilities[accepted])
            visits[max(0, left - accepted - 1)] += visits[left] * probability
            survival *= probabilities[accepted]
        visits[max(0, left - k - 1)] += visits[left] * survival
    return counts

def cache_memory(config, *, seq_len, policy='packed'):
    c = text_config(config)
    if policy not in {'packed', 'reference'} or seq_len < 0:
        raise ValueError('Invalid cache policy or sequence length')
    d, di = c['head_dim'], c['index_head_dim']
    # Main NVFP4: E2M1 + 1 E4M3/16. Index MXFP4: E2M1 + 1 E8M0/32.
    main = d/2 + math.ceil(d/16) if policy == 'packed' else d*2
    index = di/2 + math.ceil(di/32) if policy == 'packed' else di*2
    local = d + math.ceil(d/32) if policy == 'packed' else d*2
    owners = [x for x in layer_schedule(config) if x['mode'] == 'full']
    global_bytes = sum((seq_len//x['compress_ratio'])*(main+index) for x in owners)
    window_bytes = c['num_hidden_layers'] * min(seq_len, c['sliding_window']) * local
    compressor_bytes = sum(2*x['compress_ratio']*d*4 for x in owners if x['compress_ratio'] > 1)
    ed = c['engram_head_dim']
    engram_bytes = sum(c['engram_num_embeddings']) * (ed + math.ceil(ed/32))
    return dict(global_bytes=global_bytes, window_bytes=window_bytes,
        compressor_state_bytes=compressor_bytes, runtime_bytes=global_bytes+window_bytes+compressor_bytes,
        engram_table_bytes=engram_bytes, global_bytes_per_token=sum((main+index)/x['compress_ratio'] for x in owners))

def af_transfer_work(*, batch, input_len, query_len, hidden, layers, window, streams):
    """Conservative full mHC residual-stream transfer in both directions per layer."""
    values = (batch, input_len, query_len, hidden, layers, window, streams)
    if any(not isinstance(v, int) or isinstance(v, bool) or v < 1 for v in values):
        raise ValueError('AF dimensions must be positive integers')
    tokens = (layers//2)*input_len + (layers-layers//2)*min(input_len, window)
    width = hidden*streams*2*2  # BF16 residual streams, two boundaries.
    return dict(prefill_bytes=batch*tokens*width,
                verify_bytes=batch*layers*query_len*width, messages=2*layers)
